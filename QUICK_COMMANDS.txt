================================================================================
QUICK REFERENCE - COMMANDS TO RUN
================================================================================

IMPORTANT: Replace YOUR_USERNAME with your actual Docker Hub username!

================================================================================
PART 1: DOCKER HUB
================================================================================

# Build Docker image
docker build -t student-performance-api .

# Tag for Docker Hub (replace YOUR_USERNAME)
docker tag student-performance-api:latest YOUR_USERNAME/student-performance-api:latest
docker tag student-performance-api:latest YOUR_USERNAME/student-performance-api:v1.0

# Login to Docker Hub
docker login

# Push to Docker Hub
docker push YOUR_USERNAME/student-performance-api:latest
docker push YOUR_USERNAME/student-performance-api:v1.0

# Verify images
docker images | grep student-performance


================================================================================
PART 2: KUBERNETES DEPLOYMENT
================================================================================

# Before deploying, update deployment.yaml:
# Change "YOUR_USERNAME" to your actual Docker Hub username!

# Verify cluster connection
kubectl cluster-info
kubectl get nodes

# Optional: Create namespace
kubectl create namespace student-performance

# Deploy application
kubectl apply -f deployment.yaml

# Deploy service
kubectl apply -f service.yaml

# Check deployment status
kubectl get deployments
kubectl get pods
kubectl get services

# View detailed information
kubectl describe deployment student-performance-api
kubectl describe service student-performance-service

# Get pod logs
kubectl logs -l app=student-performance-api

# For Minikube - Get service URL
minikube service student-performance-service --url

# Test the API
curl http://<EXTERNAL-IP>:80/health
curl http://<EXTERNAL-IP>:80/model/info


================================================================================
OPTIONAL: CLEANUP COMMANDS
================================================================================

# Delete resources
kubectl delete -f service.yaml
kubectl delete -f deployment.yaml

# Delete namespace (if created)
kubectl delete namespace student-performance
